# Gun and saucer on the CDP1861, keys from the Elf hex pad.
#   4 left   6 right   5 fire
# It blinks the gun until you press something, so that the game does not run
# out before you have found which window takes the keyboard.
# SD is how many frames the saucer takes over each row - raise it to make the
# game easier, lower it to make it mean something.
# The machine code is assembled from asm/elfin/inline.s and printed,
# INSTALL lines and all, by ./asm/inline.sh.
INSTALL($F200, $"F8F2B1F819A1F82622527069E8F800BAF80BAAF8005A1A5AD3301D427022782252C4E2F800A0F8F1B0E2E280E2E220A0E220A0E220A03C2B301B")
INSTALL($F240, $"F8F1BDF800ADF8005D1D8D3A46E8F800BAF80BAAF8005A1A5AD3")
INSTALL($F300, $"F800BAF809AA4AAD0ABD4DAA4DBA4DAF1D4DE2FA3FACFA07BC8CF6F6F6AC8FFA1FAF4AA94AF800B9F8F1BD89327C4ABEF800AE9CBF9F32449EF6BE8E76AE9FFF01BF30358FFEFEFE528CF4AD0D529EF23255F801B99EF35D8CFC01FA07528FFEFEFEF4AD0D528EF2326DF801B98EF35D8FFC01FA1F")
INSTALL($F375, $"AF89FF01A9302BE8F800BAF80BAA995A1AF8005AD3")
INSTALL($F3A0, $"3EAF6CFF303BAFACFF0A33AF8C30B1F810ACE8F800BAF80BAA8C5A1AF8005AD3")
CLS()
VSTART()
GUN=$"18183C7EFFFFDB99"
SAU=$"183CFF7E"
BUL=$"1818"
E1=$"183C7E7E3C18"
E2=$"3C7EE7E77E3C"
E3=$"C381181881C3"
GX=28
GY=24
SX=8
SY=0
BA=0
BX=0
BY=0
SC=0
R=1
SD=2
SF=0
OUTS("PRESS A DIGIT TO START")
OUTS($"0D0A")
WHILE KEY()>9
 H=SPRITE(GX,GY,GUN)
 WAIT(150)
 H=SPRITE(GX,GY,GUN)
 WAIT(150)
H=SPRITE(GX,GY,GUN)
H=SPRITE(SX,SY,SAU)
WHILE R>0
 H=SPRITE(GX,GY,GUN)
 H=SPRITE(SX,SY,SAU)
 IF BA>0
  H=SPRITE(BX,BY,BUL)
 K=KEY()
 IF K==4
  IF GX>1
   GX=GX-2
 IF K==6
  IF GX<55
   GX=GX+2
 IF K==5
  IF BA==0
   BA=1
   BX=GX
   BY=GY-2
 IF BA>0
  IF BY<2
   BA=0
  ELSE
   BY=BY-2
 SF=SF+1
 IF SF>=SD
  SF=0
  SY=SY+1
  IF SY>23
   SY=23
   R=0
 H=SPRITE(SX,SY,SAU)
 H=SPRITE(GX,GY,GUN)
 IF BA>0
  H=SPRITE(BX,BY,BUL)
  IF H>0
   H=SPRITE(BX,BY,BUL)
   BA=0
   H=SPRITE(SX,SY,SAU)
   BOOM(SX,SY,E1,E2,E3)
   SC=SC+1
   SX=(SX*7+11)%56
   SY=0
   H=SPRITE(SX,SY,SAU)
OUTS("GAME OVER, HIT  ")
OUTI(SC)
OUTS($"0D0A")
DEF BOOM(X,Y,A[],B[],C[])
 H=SPRITE(X,Y,A)
 WAIT(120)
 H=SPRITE(X,Y,A)
 H=SPRITE(X,Y,B)
 WAIT(120)
 H=SPRITE(X,Y,B)
 H=SPRITE(X,Y,C)
 WAIT(120)
 H=SPRITE(X,Y,C)
DEF WAIT(N)
 I=0
 WHILE I<N
  I=I+1
DEF INSTALL(AT,CODE[]) $0013
DEF VSTART() $F200
DEF CLS() $F240
DEF SPRITE(X,Y,S[]) $F300
DEF KEY() $F3A0
DEF OUTS(S[]) $0017
DEF OUTI(N) $0019
