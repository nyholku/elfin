# Sierpinski on the CDP1861, driven entirely from this program.
# The three machine-code routines are assembled from
# asm/elfin/inline.s; ./asm/inline.sh prints them as these literals.
INSTALL($F200, $"F8F2B1F819A1F82622527069E8F800BAF80BAAF8005A1A5AD3301D427022782252C4E2F800A0F8F1B0E2E280E2E220A0E220A0E220A03C2B301B")
INSTALL($F240, $"F8F1BDF800ADF8005D1D8D3A46E8F800BAF80BAAF8005A1A5AD3")
INSTALL($F270, $"F800BAF809AA4AAC0ABC4CAD1C4CAEE28DFA1FFEFEFE528EFA3FF6F6F6F4ADF8F1BD8EFA07ACF880BC8C32A29CF6BC2C30999C520DF15DE8F800BAF80BAAF8005A1A5AD3")
CLS()
VSTART()
Y=0
WHILE Y<32
 X=0
 WHILE X<64
  IF X&Y==0
   PLOT(X,Y)
  X=X+1
 Y=Y+1
OUTS("DRAWN")
OUTS($"0D0A")
DEF INSTALL(AT,CODE[]) $0013
DEF VSTART() $F200
DEF CLS() $F240
DEF PLOT(X,Y) $F270
DEF OUTS(S[]) $0017
