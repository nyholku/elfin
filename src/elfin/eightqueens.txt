  
DIM BOARD[8]
SOLUTIONS=0
SOLVE(0)
OUTS("SOLUTIONS: ")
OUTI(SOLUTIONS)
OUTS($"0D0A")

DEF SOLVE(ROW)
 GLOBAL SOLUTIONS
 GLOBAL BOARD
 IF ROW==8
  SOLUTIONS=SOLUTIONS+1
  PRINT_BOARD()
  RETURN
 COL=0
 WHILE COL<8
  IF SAFE(ROW,COL)
   BOARD[ROW]=COL
   SOLVE(ROW+1)
  COL=COL+1

DEF SAFE(ROW,COL)
 GLOBAL BOARD
 I=0
 WHILE I<ROW
  IF BOARD[I]==COL
   RETURN 0
  IF BOARD[I]==COL-(ROW-I)
   RETURN 0
  IF BOARD[I]==COL+(ROW-I)
   RETURN 0
  I=I+1
 RETURN 1

DEF PRINT_BOARD()
 GLOBAL BOARD
 ROW=0
 WHILE ROW<8
  COL=0
  WHILE COL<8
   IF BOARD[ROW]==COL
    PUTCH(42)
   ELSE
    PUTCH(46)
   COL=COL+1
  PUTCH(13)
  PUTCH(10)
  ROW=ROW+1
 PUTCH(13)
 PUTCH(10)

DEF PUTCH(C) $000D
DEF OUTS(S[]) $0017
DEF OUTI(N) $0019
