# CHESS BENCHMARK in ElFin - the computing part of src/elfin/chess.txt, with
# no one to play against: the opening position, white to move, the best move
# two ply deep, and the board after it.  bench/chess.bas is the same thing in
# Netronics Tiny BASIC; bench/README.md compares the two.
#
# The engine is chess.txt's, written the way a programmer who counted cycles
# would have written it, and so is the Tiny BASIC one: the search never
# multiplies or divides.
#  - A move is two words, MVF and MVT, not FROM*256+TO.
#  - The ply lists are one stack: ply P's moves are MVF/MVT[MS[P]] up to
#    MS[P+1], so nothing is indexed by PLY*96.
#  - The position scores are tables made once by SETUP: CS for the centre,
#    PW and PK for a white and a black pawn's rank.
#  - A pawn's home rank is tested as SQ&240, the rank times 16.
DIM BRD[128]
DIM MVF[576]
DIM MVT[576]
DIM MS[9]
DIM CAP[8]
DIM VAL[7]
DIM CEN[8]
DIM CS[128]
DIM PW[128]
DIM PK[128]
NM=0
DEPTH=2
SETUP()
SHOW()
OUTS("THINKING")
OUTS($"0D0A")
MOVE=BESTMV(0,DEPTH)
SHOWMV(MVF[MOVE],MVT[MOVE])
DOMV(0,MVF[MOVE],MVT[MOVE])
SHOW()
OUTS("DONE")
OUTS($"0D0A")

DEF SETUP()
 GLOBAL BRD
 GLOBAL DIR
 GLOBAL NOFF
 GLOBAL VAL
 DIR=$"81007F00900070008F00710091006F00"
 NOFF=$"9F00A1008E00920061005F0072006E00"
 VAL[0]=0
 VAL[1]=100
 VAL[2]=320
 VAL[3]=330
 VAL[4]=500
 VAL[5]=900
 VAL[6]=20000
 GLOBAL CEN
 CEN[0]=0
 CEN[1]=1
 CEN[2]=2
 CEN[3]=3
 CEN[4]=3
 CEN[5]=2
 CEN[6]=1
 CEN[7]=0
 GLOBAL CS
 GLOBAL PW
 GLOBAL PK
 I=0
 WHILE I<128
  BRD[I]=0
  R=I/16
  CS[I]=(CEN[I&7]+CEN[R])*3
  PW[I]=R*4
  PK[I]=(7-R)*4
  I=I+1
 BACK=$"04000200030005000600030002000400"
 I=0
 WHILE I<8
  BRD[I]=BACK[I]
  BRD[16+I]=1
  BRD[96+I]=9
  BRD[112+I]=BACK[I]+8
  I=I+1
 GLOBAL MS
 MS[0]=0

DEF SHOW()
 GLOBAL BRD
 PCH=$"2E0050004E004200520051004B00"
 R=7
 WHILE R>=0
  PUTCH(49+R)
  PUTCH(32)
  F=0
  WHILE F<8
   P=BRD[R*16+F]
   C=PCH[P&7]
   IF P>8
    C=C+32
   PUTCH(C)
   PUTCH(32)
   F=F+1
  OUTS($"0D0A")
  R=R-1
 OUTS("  A B C D E F G H")
 OUTS($"0D0A")

# ---- move generation ------------------------------------------------
# GEN puts ply PLY's moves on top of the move stack, from MS[PLY], and sets
# MS[PLY+1] to where they end.  NM is the next free entry.

DEF ADDMV(FR,TO,SD)
 GLOBAL MVF
 GLOBAL MVT
 GLOBAL NM
 Q=BRD[TO]
 IF Q>0
  IF (Q&8)==SD
   RETURN
 MVF[NM]=FR
 MVT[NM]=TO
 NM=NM+1

DEF GEN(PLY,SD)
 GLOBAL BRD
 GLOBAL MS
 GLOBAL NM
 NM=MS[PLY]
 SQ=0
 WHILE SQ<128
  IF (SQ&136)==0
   P=BRD[SQ]
   IF P>0
    IF (P&8)==SD
     GENP(SQ,P,SD)
  SQ=SQ+1
 MS[PLY+1]=NM

DEF GENP(SQ,P,SD)
 T=P&7
 IF T==1
  GPAWN(SQ,SD)
 IF T==2
  GHOP(SQ,SD,1)
 IF T==3
  GSLIDE(SQ,SD,4,7)
 IF T==4
  GSLIDE(SQ,SD,0,3)
 IF T==5
  GSLIDE(SQ,SD,0,7)
 IF T==6
  GHOP(SQ,SD,0)

# One step in each direction: the king walks DIR, the knight NOFF.
DEF GHOP(SQ,SD,K)
 GLOBAL DIR
 GLOBAL NOFF
 D=0
 WHILE D<8
  IF K==0
   O=DIR[D]-128
  ELSE
   O=NOFF[D]-128
  T=SQ+O
  IF (T&136)==0
   ADDMV(SQ,T,SD)
  D=D+1

DEF GSLIDE(SQ,SD,A,B)
 GLOBAL BRD
 GLOBAL DIR
 D=A
 WHILE D<=B
  O=DIR[D]-128
  T=SQ+O
  G=1
  WHILE G==1
   IF (T&136)==0
    Q=BRD[T]
    ADDMV(SQ,T,SD)
    IF Q>0
     G=0
    T=T+O
   ELSE
    G=0
  D=D+1

# H is the home rank times 16, which is what SQ&240 gives.
DEF GPAWN(SQ,SD)
 GLOBAL BRD
 O=16
 H=16
 IF SD>0
  O=0-16
  H=96
 T=SQ+O
 IF (T&136)==0
  IF BRD[T]==0
   ADDMV(SQ,T,SD)
   IF (SQ&240)==H
    IF BRD[T+O]==0
     ADDMV(SQ,T+O,SD)
 D=0-1
 WHILE D<2
  T=SQ+O+D
  IF (T&136)==0
   IF BRD[T]>0
    ADDMV(SQ,T,SD)
  D=D+2

# ---- making and unmaking --------------------------------------------
# The captured piece is kept per ply, which is all this needs to undo: no
# castling rights, no en passant square, no promotion yet.

DEF DOMV(PLY,F,T)
 GLOBAL BRD
 GLOBAL CAP
 CAP[PLY]=BRD[T]
 BRD[T]=BRD[F]
 BRD[F]=0

DEF UNDOMV(PLY,F,T)
 GLOBAL BRD
 GLOBAL CAP
 BRD[F]=BRD[T]
 BRD[T]=CAP[PLY]

# ---- evaluation and search ------------------------------------------

# Material only, from white's point of view.  The king is worth more than
# everything else together, so a line that wins it beats any other.
DEF EVAL()
 GLOBAL BRD
 GLOBAL VAL
 S=0
 SQ=0
 WHILE SQ<128
  IF (SQ&136)==0
   P=BRD[SQ]
   IF P>0
    V=VAL[P&7]+SCORE(SQ,P)
    IF (P&8)>0
     S=S-V
    ELSE
     S=S+V
  SQ=SQ+1
 RETURN S

# What a piece is worth for standing where it does.  Material alone leaves
# every line scoring the same, and the search then plays whichever move the
# generator happened to produce first - rooks shuffling on the back rank.
DEF SCORE(SQ,P)
 T=P&7
 IF T==1
  IF (P&8)>0
   RETURN PK[SQ]
  RETURN PW[SQ]
 IF T==6
  RETURN 0
 RETURN CS[SQ]

# The same, from the side to move's point of view, which is what negamax
# wants: one routine searches for both colours because the score is always
# "good for whoever is to move".
DEF EVALS(SD)
 V=EVAL()
 IF SD>0
  RETURN 0-V
 RETURN V

DEF SEARCH(PLY,SD,D,A,B)
 GLOBAL MS
 IF D==0
  RETURN EVALS(SD)
 GEN(PLY,SD)
 I=MS[PLY]
 N=MS[PLY+1]
 BEST=0-30000
 WHILE I<N
  F=MVF[I]
  T=MVT[I]
  DOMV(PLY,F,T)
  V=0-SEARCH(PLY+1,8-SD,D-1,0-B,0-A)
  UNDOMV(PLY,F,T)
  IF V>BEST
   BEST=V
  IF BEST>A
   A=BEST
  IF A>=B
   I=N
  I=I+1
 RETURN BEST

# The index of the best move in ply 0's list.
DEF BESTMV(SD,D)
 GLOBAL MS
 GEN(0,SD)
 I=MS[0]
 N=MS[1]
 BEST=0-30000
 R=0-1
 WHILE I<N
  F=MVF[I]
  T=MVT[I]
  DOMV(0,F,T)
  V=0-SEARCH(1,8-SD,D-1,0-30000,30000)
  UNDOMV(0,F,T)
  IF V>BEST
   BEST=V
   R=I
  I=I+1
 RETURN R

DEF SHOWMV(F,T)
 PUTCH(97+(F&7))
 PUTCH(49+(F/16))
 PUTCH(97+(T&7))
 PUTCH(49+(T/16))
 OUTS($"0D0A")

DEF PUTCH(C) $000D
DEF GETCH() $000F
DEF OUTS(S[]) $0017
DEF OUTI(N) $0019
